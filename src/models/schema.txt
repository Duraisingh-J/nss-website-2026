2. people
people
──────────────────────────────
id                    uuid            PK
name                  text            NOT NULL
designation           text            NOT NULL            
role_id               uuid            NOT NULL, FK -> roles.id   ON DELETE RESTRICT
registration_number   text            NULL UNIQUE
department            text            NOT NULL
year                  smallint        NULL for PO (CHECK year IS NULL OR year BETWEEN 1 AND 4)
unit                  smallint        NOT NULL (CHECK unit BETWEEN 1 AND 7)
phone                 text            NOT NULL
email                 text            NOT NULL
photo_media_id        uuid            NULL for PO (maybe)   ON DELETE SET NULL
bio                   text            NULL for PO 
is_active             boolean         DEFAULT true
created_at            timestamptz     NOT NULL DEFAULT now()
updated_at            timestamptz     NOT NULL DEFAULT now()

3. Media
media
────────────────────────────
id            uuid            PK    
file_name     text            NOT NULL UNIQUE
storage_path  text            NOT NULL UNIQUE
mime_type     text            NOT NULL
file_size     bigint          NOT NULL (CHECK file_size > 0)
width         integer         NULLABLE for PDF (CHECK width is NULL OR width > 0)
height        integer         NULLABLE for PDF (CHECK height is NULL OR height > 0)
alt_text      text            NULLABLE
caption       text            NULLABLE
created_at    timestamptz     NOT NULL DEFAULT now()
updated_at    timestamptz     NOT NULL DEFAULT now()

4. Events
events
────────────────────────────
id              uuid        PK  
title           text        NOT NULL
description     text        NULL
event_type      text        NOT NULL
start_date      date        NOT NULL 
end_date        date        NOT NULL (CHECK end_date >= start_date)
cover_media_id  uuid        NULLABLE, FK -> media.id ON DELETE SET NULL
is_published    boolean     DEFAULT false
created_at      timestamptz     NOT NULL DEFAULT now()
updated_at      timestamptz     NOT NULL DEFAULT now()

event_type IN (
  'camp',
  'visit',
  'drive',
  'awareness',
  'workshop',
  'other'
)
